        function playLaserSound(isEnemy = false) {
            if (audioCtx.state === 'suspended') audioCtx.resume();
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(isEnemy ? 400 : 800, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.15);
            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.15);
        }

        function initThree() {
            const container = document.getElementById('canvas-container');
            scene = new THREE.Scene();
            scene.background = new THREE.Color(0x0a0c10);
            scene.fog = new THREE.FogExp2(0x0a0c10, 0.015);

            camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
            renderer = new THREE.WebGLRenderer({ antialias: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.shadowMap.enabled = true;
            renderer.shadowMap.type = THREE.PCFSoftShadowMap;
            container.appendChild(renderer.domElement);

            clock = new THREE.Clock();

            // Lighting
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.4);
            scene.add(ambientLight);

            const dirLight = new THREE.DirectionalLight(0xffeedd, 0.8);
            dirLight.position.set(50, 100, 50);
            dirLight.castShadow = true;
            dirLight.shadow.mapSize.width = 2048;
            dirLight.shadow.mapSize.height = 2048;
            scene.add(dirLight);

            // Environment Floor
            const floorGeo = new THREE.PlaneGeometry(200, 200);
            const floorMat = new THREE.MeshStandardMaterial({ color: 0x1e222b, roughness: 0.8 });
            const floor = new THREE.Mesh(floorGeo, floorMat);
            floor.rotation.x = -Math.PI / 2;
            floor.receiveShadow = true;
            scene.add(floor);

            // Outpost Obstacles
            const boxGeo = new THREE.BoxGeometry(4, 4, 4);
            const boxMat = new THREE.MeshStandardMaterial({ color: 0x3a3f4d });
            for (let i = 0; i < 20; i++) {
                const box = new THREE.Mesh(boxGeo, boxMat);
                box.position.set((Math.random() - 0.5) * 120, 2, (Math.random() - 0.5) * 120);
                box.castShadow = true;
                box.receiveShadow = true;
                scene.add(box);
            }

            // Player Group Setup
            player = new THREE.Group();
            scene.add(player);
            player.position.set(0, 0, 0);

            buildPlayerModel();
            spawnDroids(6);

            window.addEventListener('resize', onWindowResize);
            setupInput();
        }

        // Detailed Player Model Construction
        function buildPlayerModel() {
            if (playerBody) player.remove(playerBody);
            playerBody = new THREE.Group();

            const mat = new THREE.MeshStandardMaterial({ color: CHARACTERS[selectedChar].color, roughness: 0.5 });

            const torso = new THREE.Mesh(new THREE.BoxGeometry(0.8, 1.2, 0.5), mat);
            torso.position.y = 1.4;
            torso.castShadow = true;
            playerBody.add(torso);

            // Helmet
            const head = new THREE.Mesh(new THREE.SphereGeometry(0.35, 16, 16), mat);
            head.position.y = 2.2;
            head.castShadow = true;
            
            // Visor
            const visor = new THREE.Mesh(new THREE.BoxGeometry(0.4, 0.12, 0.2), new THREE.MeshBasicMaterial({ color: 0x00e5ff }));
            visor.position.set(0, 2.22, 0.25);
            playerBody.add(visor);
            playerBody.add(head);

            // 773 Firepunch Rifle Mesh for Crosshair / Generic Weapons
            weaponGroup = new THREE.Group();
            const bodyGeo = new THREE.BoxGeometry(0.1, 0.15, 1.2);
            const bodyMesh = new THREE.Mesh(bodyGeo, armorMat);
            
            const barrelGeo = new THREE.CylinderGeometry(0.03, 0.03, 0.8);

            weaponGroup.position.set(0.4, 1.3, -0.5);
            player.add(weaponGroup);

            player.add(playerBody);
        }

        // B1 Battle Droid Mesh Generator based on references
        function createB1Droid() {
            const droid = new THREE.Group();
            const tanMat = new THREE.MeshStandardMaterial({ color: 0xc2a675, roughness: 0.6 });
            const darkMat = new THREE.MeshStandardMaterial({ color: 0x333333 });

            // Head (Cone structure)
            const head = new THREE.Mesh(new THREE.ConeGeometry(0.15, 0.5, 8), tanMat);
            head.rotation.x = Math.PI / 3;
            head.position.set(0, 1.7, 0.1);
            droid.add(head);

            // Torso & Backpack
            const torso = new THREE.Mesh(new THREE.BoxGeometry(0.3, 0.6, 0.2), tanMat);
            torso.position.y = 1.2;
            
            const pack = new THREE.Mesh(new THREE.BoxGeometry(0.35, 0.4, 0.25), tanMat);
            pack.position.set(0, 1.3, -0.2);

            const antenna = new THREE.Mesh(new THREE.CylinderGeometry(0.01, 0.01, 0.8), darkMat);
            antenna.position.set(0.12, 1.7, -0.3);
            
            droid.add(torso, pack, antenna);

            // Thin Limbs
            const legGeo = new THREE.CylinderGeometry(0.04, 0.04, 0.8);
            const leftLeg = new THREE.Mesh(legGeo, tanMat);
            leftLeg.position.set(-0.15, 0.4, 0);
            const rightLeg = new THREE.Mesh(legGeo, tanMat);
            rightLeg.position.set(0.15, 0.4, 0);
            droid.add(leftLeg, rightLeg);

            droid.castShadow = true;
            return droid;
        }

        function spawnDroids(count) {
            for (let i = 0; i < count; i++) {
                const d = createB1Droid();
                d.position.set((Math.random() - 0.5) * 80, 0, (Math.random() - 0.5) * 80);
                d.userData = { hp: 50, lastShoot: 0 };
                scene.add(d);
                droids.push(d);
            }
        }

        function setupInput() {
            document.addEventListener('keydown', (e) => {
                if (e.key.toLowerCase() in keys) keys[e.key.toLowerCase()] = true;
                if (e.key === ' ') keys.space = true;
                if (e.shiftKey) keys.shift = true;
                if (e.key.toLowerCase() === 'v') toggleView();
            });

            document.addEventListener('keyup', (e) => {
                if (e.key.toLowerCase() in keys) keys[e.key.toLowerCase()] = false;
                if (e.key === ' ') keys.space = false;
            });

            document.addEventListener('mousedown', (e) => {
                if (!isGameRunning) return;
                if (e.button === 0) shoot();
                if (e.button === 2) setADS(true);
            });

            document.addEventListener('mouseup', (e) => {
                if (e.button === 2) setADS(false);
            });

            document.addEventListener('contextmenu', e => e.preventDefault());

            document.addEventListener('mousemove', (e) => {
                if (document.pointerLockElement === document.body) {
                    const sens = isADS ? 0.001 : 0.0025;
                    yaw -= e.movementX * sens;
                    pitch -= e.movementY * sens;
                    pitch = Math.max(-Math.PI / 2.2, Math.min(Math.PI / 2.2, pitch));
                }
            });
        }

        function toggleView() {
            isTPP = !isTPP;
            playerBody.visible = isTPP;
        }

        function setADS(active) {
            isADS = active;
            const scopeEl = document.getElementById('sniper-scope');
            const crosshairEl = document.getElementById('crosshair');

            if (selectedChar === 'crosshair' && isADS) {
                scopeEl.style.display = 'block';
                crosshairEl.style.display = 'none';
                weaponGroup.visible = false;
                if (!isTPP) playerBody.visible = false;
            } else {
                scopeEl.style.display = 'none';
                crosshairEl.style.display = 'block';
                weaponGroup.visible = true;
            }
        }

        function shoot() {
            if (ammo <= 0) return;
            ammo--;
            document.getElementById('ammo-count').innerText = ammo;
            playLaserSound(false);

            // Create Player Laser Bolt
            const laserGeo = new THREE.CylinderGeometry(0.03, 0.03, 1);
            const laserMat = new THREE.MeshBasicMaterial({ color: 0x00aaff });
            const laser = new THREE.Mesh(laserGeo, laserMat);
            laser.rotation.x = Math.PI / 2;

            const dir = new THREE.Vector3();
            camera.getWorldDirection(dir);
            laser.position.copy(camera.position).add(dir.clone().multiplyScalar(1));
            laser.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, -1), dir);

            scene.add(laser);
            lasers.push({ mesh: laser, dir: dir, life: 1.5, isEnemy: false });

            // Raycast Hit Detection
            const raycaster = new THREE.Raycaster(camera.position, dir);
            const hits = raycaster.intersectObjects(droids, true);

            if (hits.length > 0) {
                let hitDroid = hits[0].object;
                while (hitDroid.parent && hitDroid.parent !== scene) hitDroid = hitDroid.parent;

                hitDroid.userData.hp -= CHARACTERS[selectedChar].dmg;
                if (hitDroid.userData.hp <= 0) {
                    scene.remove(hitDroid);
                    droids = droids.filter(d => d !== hitDroid);
                    setTimeout(() => spawnDroids(1), 3000); // Respawn
                }
            }
        }

        function startGame() {
            document.getElementById('lobby-screen').style.display = 'none';
            document.getElementById('hud').style.display = 'block';
            document.body.requestPointerLock();
            
            ammo = CHARACTERS[selectedChar].maxAmmo;
            document.getElementById('ammo-max').innerText = ammo;
            document.getElementById('ammo-count').innerText = ammo;
            
            buildPlayerModel();
            isGameRunning = true;
            
            if (!clock.running) {
                initThree();
                animate();
            }
        }

        function restartGame() {
            document.getElementById('game-over').style.display = 'none';
            document.getElementById('lobby-screen').style.display = 'flex';
            hp = 100;
            document.getElementById('hp-fill').style.width = '100%';
        }

        function animate() {
            requestAnimationFrame(animate);
            const delta = clock.getDelta();

            if (isGameRunning) {
                // Update Player Orientation
                player.rotation.y = yaw;
                camera.rotation.x = pitch;

                // Physics & Movement
                const moveDir = new THREE.Vector3();
                if (keys.w) moveDir.z -= 1;
                if (keys.s) moveDir.z += 1;
                if (keys.a) moveDir.x -= 1;
                if (keys.d) moveDir.x += 1;
                moveDir.normalize();

                const currentSpeed = keys.shift ? moveSpeed * 1.6 : moveSpeed;
                player.translateX(moveDir.x * currentSpeed * delta);
                player.translateZ(moveDir.z * currentSpeed * delta);

                // Camera Position Attachment
                if (isTPP) {
                    const offset = new THREE.Vector3(0, 2.2, 4).applyAxisAngle(new THREE.Vector3(0, 1, 0), yaw);
                    camera.position.copy(player.position).add(offset);
                    camera.rotation.y = yaw;
                } else {
                    camera.position.copy(player.position).add(new THREE.Vector3(0, 1.8, 0));
                    camera.rotation.y = yaw;
                }

                // Camera FOV Zoom for ADS
                const targetFOV = isADS ? (selectedChar === 'crosshair' ? 20 : 45) : 75;
                camera.fov = THREE.MathUtils.lerp(camera.fov, targetFOV, delta * 12);
                camera.updateProjectionMatrix();

                // Update Lasers
                for (let i = lasers.length - 1; i >= 0; i--) {
                    const l = lasers[i];
                    l.mesh.position.add(l.dir.clone().multiplyScalar(60 * delta));
                    l.life -= delta;
                    if (l.life <= 0) {
                        scene.remove(l.mesh);
                        lasers.splice(i, 1);
                    }
                }

                // B1 Droid AI Logic
                droids.forEach(droid => {
                    const dist = droid.position.distanceTo(player.position);
                    droid.lookAt(player.position.x, droid.position.y, player.position.z);

                    if (dist > 5 && dist < 40) {
                        droid.translateZ(3 * delta);
                    }

                    // Droid Fire Rate Loop
                    if (dist < 30 && performance.now() - droid.userData.lastShoot > 2000) {
                        droid.userData.lastShoot = performance.now();
                        playLaserSound(true);

                        const laserGeo = new THREE.CylinderGeometry(0.03, 0.03, 1);
                        const laserMat = new THREE.MeshBasicMaterial({ color: 0xff0000 });
                        const laser = new THREE.Mesh(laserGeo, laserMat);
                        laser.rotation.x = Math.PI / 2;

                        const dir = new THREE.Vector3().subVectors(player.position, droid.position).normalize();
                        laser.position.copy(droid.position).add(new THREE.Vector3(0, 1.4, 0));
                        laser.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, -1), dir);

                        scene.add(laser);
                        lasers.push({ mesh: laser, dir: dir, life: 1.5, isEnemy: true });

                        // Damage Player
                        if (dist < 25) {
                            hp -= 8;
                            document.getElementById('hp-fill').style.width = Math.max(0, hp) + '%';
                            if (hp <= 0) {
                                isGameRunning = false;
                                document.exitPointerLock();
                                document.getElementById('game-over').style.display = 'flex';
                            }
                        }
                    }
                });
            }

            renderer.render(scene, camera);
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }
